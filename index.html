async function fetchFDJResults() { state.loadingFDJ = true; render(); try { // Appeler le backend Render const response = await fetch('https://lotopotes.onrender.com/api/loto-results'); if (response.ok) { const data = await response.json(); state.fdjResults = data.results || getMockFDJResults(); console.log('✅ Données FDJ récupérées depuis le backend !'); } else { console.log('Backend indisponible, utilisation des données simulées'); state.fdjResults = getMockFDJResults(); } } catch (e) { console.log('Erreur backend:', e.message); console.log('Utilisation des données simulées FDJ'); state.fdjResults = getMockFDJResults(); } state.loadingFDJ = false; render(); }
